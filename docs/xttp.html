xttp

Fiber v3 routes that also describe themselves as an OpenAPI 3.1 document. It looks like Huma from the outside, one struct per operation with OpenAPI from Go types, but it does not wrap your handler: handlers stay plain fiber.Handler.

  • Production server by default: xttp.New(Config) builds the Fiber app with timeouts, body limit, rate limit, probes, request id, access log, recovery and security headers
  • Plain handlers: func(c fiber.Ctx) error, no input/output wrapper structs
  • Schemas from Go types: structs, pointers, slices, maps, time.Time, []byte, embedded and recursive types; named structs become components/schemas
  • Parameters: Fiber :id / :id? become path params; a Query struct's query:"name" fields become query params
  • JSON:API errors with no internal text on the wire; Bind / BindQuery return them as 400s
  • Auth helpers: constant-time BearerTokens and APIKey, Principal, documented via Route.Security
  • Startup-only reflection; no YAML dependency

Install

go get github.com/moeghifar/libgo/pkg/xttp github.com/gofiber/fiber/v3

Quick start

package main

import (
	"context"

	"github.com/gofiber/fiber/v3"
	"github.com/moeghifar/libgo/pkg/xttp"
)

type User struct {
	ID    int    `json:"id"`
	Name  string `json:"name" doc:"Full name"`
	Email string `json:"email,omitempty"`
}

type ListUsers struct {
	Page int    `query:"page" doc:"Page number"`
	Sort string `query:"sort" required:"true"` // enforced: missing → 400
}

var ErrBadInput = xttp.Error{Status: 400, Code: "USR-400", Title: "Invalid input"}

func main() {
	api := xttp.New(xttp.Config{
		Info: xttp.Info{Title: "Users", Version: "1.0.0"},
		Docs: xttp.DocsConfig{Enabled: true}, // /docs, /docs/openapi.yaml, /docs/openapi.json
	})
	v1 := api.Group("/v1")

	v1.Add(
		xttp.Route{
			Method:    fiber.MethodGet,
			Path:      "/users",
			Summary:   "List users",
			Tags:      []string{"users"},
			Query:     ListUsers{},
			Responses: map[int]any{200: []User{}},
			Handler: func(c fiber.Ctx) error {
				var q ListUsers
				if err := xttp.BindQuery(c, &q); err != nil {
					return err // already a 400 *xttp.Error
				}
				return c.JSON([]User{})
			},
		},
		xttp.Route{
			Method:    fiber.MethodPost,
			Path:      "/users",
			Summary:   "Create a user",
			Tags:      []string{"users"},
			Body:      User{},
			Responses: map[int]any{201: User{}},
			Handler: func(c fiber.Ctx) error {
				var u User
				if err := xttp.Bind(c, &u); err != nil { // strict JSON: unknown fields → 400
					return err
				}
				return c.Status(fiber.StatusCreated).JSON(u)
			},
		},
	)

	api.Run(context.Background(), ":3000") // graceful stop on SIGINT/SIGTERM
}

type Route struct {
	Method      string // fiber.MethodGet, ...
	Path        string // Fiber syntax: "/users/:id"; ":id?" marks an optional param
	Summary     string
	Description string
	OperationID string
	Tags        []string
	Deprecated  bool

	Query     any           // struct with query:"name" fields
	Body      any           // request body type
	Responses map[int]any   // status → body type; ≥400 with nil → ErrorDocument
	Security  []string      // names from Config.SecuritySchemes

	Handler     fiber.Handler
	Middlewares []fiber.Handler // run before Handler, on this route only
}

type Error struct {
	Status int            // HTTP status
	Code   string         // application-specific code, e.g. "USR-404"
	Title  string         // short summary, the same for every occurrence
	Detail string         // explanation of this occurrence, shown to the client
	Source *ErrorSource   // what caused it in the request
	Meta   map[string]any
	Err    error          // internal cause: logged, never sent to the client
}

Middleware order: probes → request id → access log → recover → security headers → CORS → rate limit → pprof → content-type check → docs → your routes.

Errors (JSON:API)

Every error leaves as a JSON:API error document, Content-Type: application/vnd.api+json. With returns a copy; only *Error is an error, so a shared value cannot be mutated by accident.

var ErrUserNotFound = xttp.Error{Status: 404, Code: "USR-404", Title: "User not found"}

func getUser(c fiber.Ctx) error {
	u, err := repo.Find(c.Params("id"))
	if err != nil {
		return ErrUserNotFound.With("no user with id "+c.Params("id"), err) // err is logged, never sent
	}
	return c.JSON(u)
}
{"errors":[{"id":"<request id>","status":"404","code":"USR-404","title":"User not found","detail":"no user with id 42"}]}
  • xttp.Errors{...} sends several at once (validation)
  • Fiber errors (404, 405, 413) keep their status; any other error is a 500 with no internal text, logged with slog
  • Cleanup on shutdown: api.App().Hooks().OnPostShutdown(func(err error) error { return db.Close() }), idempotent

Real usage: the hardened server from example/

// newAPI is the hardened server every service starts from.
func newAPI(cfg ServeConfig) *xttp.API {
	return xttp.New(xttp.Config{
		Info:              xttp.Info{Title: "Example", Version: version},
		Fiber:             fiber.Config{TrustProxy: cfg.TrustProxy},
		HelmetPreset:      xttp.HelmetStrict,
		StrictContentType: true,
		RateLimit:         cfg.RateLimit,
		DefaultErrors:     []int{400, 401, 404, 415, 429, 500},
		SecuritySchemes:   map[string]any{"bearer": map[string]any{"type": "http", "scheme": "bearer"}},
		Docs:              xttp.DocsConfig{Enabled: cfg.Docs},
		Listen:            fiber.ListenConfig{DisableStartupMessage: true},
	})
}

// a protected group: constant-time token check, documented as "bearer"
api.Group("/v1", xttp.BearerTokens(cfg.APITokens...)).Add(user.Routes(welcome.Submit)...)

And a route in example/internal/user/user.go that binds strictly, returns a typed error, and queues a job:

var ErrInvalid = xttp.Error{Status: fiber.StatusBadRequest, Code: "USR-400", Title: "Invalid user"}

func Routes(submit func(context.Context, WelcomeJob) error) []xttp.Route {
	return []xttp.Route{{
		Method:    fiber.MethodPost,
		Path:      "/users",
		Summary:   "Create a user",
		Tags:      []string{"users"},
		Body:      User{},
		Security:  []string{"bearer"},
		Responses: map[int]any{fiber.StatusCreated: User{}},
		Handler: func(c fiber.Ctx) error {
			var u User
			if err := xttp.Bind(c, &u); err != nil {
				return err
			}
			if u.Name == "" || u.Email == "" {
				return ErrInvalid.With("name and email are required", nil)
			}
			u.ID = 1
			if err := submit(c.Context(), WelcomeJob{Email: u.Email}); err != nil {
				return err // pool shutting down: a 500, logged, details hidden
			}
			return c.Status(fiber.StatusCreated).JSON(u)
		},
	}}
}
