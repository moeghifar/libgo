climd

A command-line interface builder designed as a simpler alternative to Cobra. One AppConfig literal describes the whole app; stdlib only.

  • Commands, subcommands and flags from a single struct
  • ./cmd <command> --flag value --bool-flag and ./cmd <command> <subcommand> [flags]
  • Context-aware Run functions
  • Built-in --help and --version
  • Default command run when no arguments are given; otherwise help is printed

Install

go get github.com/moeghifar/libgo/pkg/climd

Quick start

package main

import (
	"context"
	"fmt"

	"github.com/moeghifar/libgo/pkg/climd"
)

func main() {
	climd.Execute(climd.AppConfig{
		Name:        "myapp",
		Version:     "1.0.0",
		Description: "A sample CLI application",
		Default:     "serve", // run with no arguments
		Commands: []climd.Command{
			{
				Name:  "serve",
				Short: "Start the server",
				Flags: []climd.Flag{
					{Name: "http", Usage: "HTTP modules (all, none, a,b)"},
					{Name: "grpc", Usage: "gRPC modules"},
				},
				Run: func(ctx context.Context, args []string, flags map[string]string) error {
					fmt.Printf("HTTP: %s, gRPC: %s\n", flags["http"], flags["grpc"])
					return nil
				},
			},
			{
				Name:  "db",
				Short: "Database operations",
				SubCommands: []climd.SubCommand{
					{Name: "init", Short: "Initialize the database",
						Run: func(ctx context.Context, _ []string, _ map[string]string) error {
							fmt.Println("init")
							return nil
						}},
					{Name: "create_sql", Short: "Generate SQL files",
						Flags: []climd.Flag{{Name: "output", Short: "o", Usage: "Output directory"}},
						Run: func(ctx context.Context, _ []string, flags map[string]string) error {
							fmt.Println("writing to", flags["output"])
							return nil
						}},
				},
			},
		},
	})
}
./myapp serve --http=all --grpc=module1
./myapp db init
./myapp db create_sql --output=./migrations
./myapp --help

API

SymbolMeaning
func Execute(config AppConfig)Runs the app with os.Args
func Run(config AppConfig, args []string) errorRuns with explicit arguments; returns the command's error (use it in tests)
type AppConfig struct {
	Name        string
	Version     string
	Description string
	Commands    []Command
	Default     string // command run when no arguments are given; empty prints help
}

type Command struct {
	Name        string
	Short       string
	Long        string
	Flags       []Flag
	SubCommands []SubCommand
	Run         func(ctx context.Context, args []string, flags map[string]string) error
}

type SubCommand struct {
	Name  string
	Short string
	Long  string
	Flags []Flag
	Run   func(ctx context.Context, args []string, flags map[string]string) error
}

type Flag struct {
	Name     string
	Short    string
	Value    string // default
	Usage    string
	Required bool
}

A boolean flag (--migrate with no value) shows up as a key in flags; check it with _, ok := flags["migrate"].

Real usage: typed config per command

From example/main.go: a generic run adapter loads the command's own config with envy, installs a signal context and calls the typed function. The Run signature stays the plain climd one.

// run wires every command: config for that command, and a context
// cancelled on SIGINT or SIGTERM.
func run[C any](fn func(context.Context, C) error) func(context.Context, []string, map[string]string) error {
	return func(ctx context.Context, _ []string, _ map[string]string) error {
		var cfg C
		if err := envy.Load(&cfg); err != nil {
			return fmt.Errorf("config: %w", err)
		}
		ctx, stop := signal.NotifyContext(ctx, os.Interrupt, syscall.SIGTERM)
		defer stop()
		return fn(ctx, cfg)
	}
}

func main() {
	climd.Execute(climd.AppConfig{
		Name:        "example",
		Version:     version, // -ldflags "-X main.version=1.2.3"
		Description: "Reference libgo service",
		Commands: []climd.Command{
			{Name: "serve", Short: "Run the HTTP API and workers", Run: run(serve)},
			{Name: "migrate", Short: "Run database migrations", Run: run(migrate)},
		},
	})
}
