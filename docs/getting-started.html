Getting started

example/ is the reference bootstrap: copy it to start a new service. Four packages, one binary, two commands.

Run it

git clone https://github.com/moeghifar/libgo && cd libgo
cp example/.env.example example/.env
echo "API_TOKENS=$(openssl rand -hex 32)" >> example/.env
echo "DOCS_ENABLED=true" >> example/.env

go run ./example serve
# in another shell
TOKEN=$(grep API_TOKENS example/.env | tail -1 | cut -d= -f2)
curl -X POST localhost:8080/v1/users -H 'Content-Type: application/json' \
  -H "Authorization: Bearer $TOKEN" -d '{"name":"Ada","email":"ada@example.com"}'
open http://localhost:8080/docs

Layout

example/
├── main.go          climd commands; run[C] loads config + signal context
├── config.go        envy structs: Log, ServeConfig, MigrateConfig
├── serve.go         xttp server + foreverwork pool, wired and drained in order
├── migrate.go       slot for one-shot jobs
├── internal/user/   a sample business module: types, routes, jobs, tests
├── .env.example     every setting with a comment
└── Dockerfile       hardened multi-stage image

1. Commands and config

Each command loads only its own config. The generic run adapter turns a typed func(ctx, Config) error into the climd signature and installs a context cancelled on SIGINT/SIGTERM.

var version = "dev" // -ldflags "-X main.version=1.2.3"

func run[C any](fn func(context.Context, C) error) func(context.Context, []string, map[string]string) error {
	return func(ctx context.Context, _ []string, _ map[string]string) error {
		var cfg C
		if err := envy.Load(&cfg); err != nil {
			return fmt.Errorf("config: %w", err)
		}
		ctx, stop := signal.NotifyContext(ctx, os.Interrupt, syscall.SIGTERM)
		defer stop()
		return fn(ctx, cfg)
	}
}

func main() {
	climd.Execute(climd.AppConfig{
		Name: "example", Version: version, Description: "Reference libgo service",
		Commands: []climd.Command{
			{Name: "serve", Short: "Run the HTTP API and workers", Run: run(serve)},
			{Name: "migrate", Short: "Run database migrations", Run: run(migrate)},
		},
	})
}
type Log struct {
	Level string `env:"LOG_LEVEL" default:"INFO"`
}

// setup makes JSON on stdout the default logger.
func (l Log) setup() error {
	var level slog.Level
	if err := level.UnmarshalText([]byte(l.Level)); err != nil {
		return fmt.Errorf("LOG_LEVEL: %w", err)
	}
	slog.SetDefault(slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{Level: level})))
	return nil
}

type ServeConfig struct {
	Log
	APITokens  []string `env:"API_TOKENS" required:"true"`
	Addr       string   `env:"APP_ADDR" default:"127.0.0.1:8080"`
	RateLimit  int      `env:"RATE_LIMIT" default:"100"`
	Docs       bool     `env:"DOCS_ENABLED" default:"false"`
	TrustProxy bool     `env:"TRUST_PROXY" default:"false"`
}

2. Server and workers

The pool starts first and stops last: the API drains before the pool, so no request submits to a stopped pool.

func serve(ctx context.Context, cfg ServeConfig) error {
	if err := cfg.setup(); err != nil {
		return err
	}
	ctx, cancel := context.WithCancel(ctx)
	defer cancel()

	wp := foreverwork.New()
	welcome := foreverwork.Register(wp,
		foreverwork.Config{Name: "welcome", Workers: 2, QueueSize: 10, Timeout: 5 * time.Second},
		user.SendWelcome)
	workers := make(chan struct{})
	go func() { wp.Run(ctx); close(workers) }()

	api := newAPI(cfg)
	api.Group("/v1", xttp.BearerTokens(cfg.APITokens...)).Add(user.Routes(welcome.Submit)...)

	slog.InfoContext(ctx, "starting", "version", version, "addr", cfg.Addr)
	err := api.Run(ctx, cfg.Addr) // returns after the signal and the drain
	cancel()
	<-workers
	return err
}

func newAPI(cfg ServeConfig) *xttp.API {
	return xttp.New(xttp.Config{
		Info:              xttp.Info{Title: "Example", Version: version},
		Fiber:             fiber.Config{TrustProxy: cfg.TrustProxy},
		HelmetPreset:      xttp.HelmetStrict,
		StrictContentType: true,
		RateLimit:         cfg.RateLimit,
		DefaultErrors:     []int{400, 401, 404, 415, 429, 500},
		SecuritySchemes:   map[string]any{"bearer": map[string]any{"type": "http", "scheme": "bearer"}},
		Docs:              xttp.DocsConfig{Enabled: cfg.Docs},
		Listen:            fiber.ListenConfig{DisableStartupMessage: true},
	})
}

3. A business module

internal/user owns its types, routes and jobs. It depends on the pool only through a func, so the test passes a counter.

package user

type User struct {
	ID    int    `json:"id,omitempty"`
	Name  string `json:"name" doc:"Full name"`
	Email string `json:"email"`
}

type WelcomeJob struct{ Email string }

var ErrInvalid = xttp.Error{Status: fiber.StatusBadRequest, Code: "USR-400", Title: "Invalid user"}

func SendWelcome(ctx context.Context, _ WelcomeJob) error {
	slog.InfoContext(ctx, "welcome email sent")
	return nil
}

func Routes(submit func(context.Context, WelcomeJob) error) []xttp.Route {
	return []xttp.Route{{
		Method:    fiber.MethodPost,
		Path:      "/users",
		Summary:   "Create a user",
		Tags:      []string{"users"},
		Body:      User{},
		Security:  []string{"bearer"},
		Responses: map[int]any{fiber.StatusCreated: User{}},
		Handler: func(c fiber.Ctx) error {
			var u User
			if err := xttp.Bind(c, &u); err != nil {
				return err
			}
			if u.Name == "" || u.Email == "" {
				return ErrInvalid.With("name and email are required", nil)
			}
			u.ID = 1
			if err := submit(c.Context(), WelcomeJob{Email: u.Email}); err != nil {
				return err
			}
			return c.Status(fiber.StatusCreated).JSON(u)
		},
	}}
}
func TestCreate(t *testing.T) {
	tests := []struct {
		name   string
		body   string
		status int
		jobs   int
	}{
		{"valid", `{"name":"Ada","email":"ada@example.com"}`, 201, 1},
		{"missing email", `{"name":"Ada"}`, 400, 0},
		{"not json", `nope`, 400, 0},
		{"unknown field", `{"name":"A","email":"a@b.c","admin":true}`, 400, 0},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			jobs := 0
			api := xttp.New(xttp.Config{SecuritySchemes: map[string]any{"bearer": map[string]any{"type": "http", "scheme": "bearer"}}})
			api.Group("/v1").Add(Routes(func(context.Context, WelcomeJob) error { jobs++; return nil })...)

			req := httptest.NewRequest("POST", "/v1/users", strings.NewReader(tt.body))
			req.Header.Set("Content-Type", "application/json")
			resp, err := api.App().Test(req)
			if err != nil {
				t.Fatal(err)
			}
			if resp.StatusCode != tt.status || jobs != tt.jobs {
				t.Fatalf("got status %d, jobs %d; want %d, %d", resp.StatusCode, jobs, tt.status, tt.jobs)
			}
		})
	}
}

4. Ship it

make check                      # vet, race tests, lint, security, vuln, trivy, gitleaks
make docker && make docker-run  # hardened image on 127.0.0.1:8080
go build -tags libgo_envy_slim -ldflags "-X main.version=1.2.3" ./example

What you get for free from the hardened profile: request ID, slog access log, panic recovery, strict helmet headers and Cache-Control: no-store on every response, no Server banner, per-IP rate limit, 1 MiB body limit, timeouts, dual-stack listener, Kubernetes probes, JSON:API errors, 415 on non-JSON bodies, constant-time bearer auth.
